from more_common import *

def run(r,options):
    mode=options['mode'];d=r.new()
    r.sketch(d,draw=lambda sm:sm.CreateCornerRectangle(.01,0.,0.,.03,.03,0.))
    extrude(r,d,.04,'Rufa_Base')
    d.ClearSelection2(True)
    b=r.bodies(d)[0]
    if mode=='mirror':
        r.check('mirror_plane_body',d.FeatureByName('右视基准面').Select2(False,2) and sel(d,b,256,True))
        f=r.feature(d.FeatureManager.InsertMirrorFeature2(True,False,False,False,0),'Rufa_BodyMirror')
        target=[[-30.,0.,0.,-10.,30.,40.],[10.,0.,0.,30.,30.,40.]]
    elif mode=='copy':
        r.check('copy_body_selected',sel(d,b,1))
        f=r.feature(d.FeatureManager.InsertMoveCopyBody2(.05,0.,0.,0.,0.,0.,0.,0.,0.,0.,True,1),'Rufa_BodyCopy')
        target=[[10.,0.,0.,30.,30.,40.],[60.,0.,0.,80.,30.,40.]]
    else:
        r.check('rotate_body_selected',sel(d,b,1))
        f=r.feature(d.FeatureManager.InsertMoveCopyBody2(0.,0.,0.,0.,0.,0.,0.,0.,0.,math.pi/2,False,1),'Rufa_BodyRotate')
        target=[[-30.,10.,0.,0.,30.,40.]]
    def verify(doc):
        boxes=sorted([r.box(b) for b in r.bodies(doc)])
        r.check('actual_body_positions',len(boxes)==len(target) and all(max(abs(x-y) for x,y in zip(a,b))<.001 for a,b in zip(boxes,target)),boxes_mm=boxes)
        r.check('actual_body_volume',abs(r.volume(doc)-24000*len(target))<.01)
    verify(d)
    p=r.save(d,mode+'.SLDPRT');d=r.reopen(d,p);verify(d);r.preview(d,mode)
