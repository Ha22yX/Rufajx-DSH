r=MoreRun('explode-bom');error=None
def as_interface(obj,name):
    lib=LoadTypeLibEx('D:/Program Files/Solidworks/SOLIDWORKS/sldworks.tlb',REGKIND_NONE)
    for i in range(lib.GetTypeInfoCount()):
        if lib.GetDocumentation(i)[0]==name:
            guid=str(lib.GetTypeInfo(i).GetTypeAttr().guid)
            return win32com.client.Dispatch(obj._oleobj_.QueryInterface(pythoncom.MakeIID(guid),pythoncom.IID_IDispatch))
    raise RuntimeError(name)
def pos(c):
    return list(get(c.GetTotalTransform(True),'ArrayData')[9:12])
def table_data(bom,config):
    table=as_interface(bom,'ITableAnnotation');table._FlagAsMethod('Text')
    cells=[[table.Text(i,j) for j in range(table.ColumnCount)] for i in range(table.RowCount)]
    counts=[]
    for row in range(table.RowCount):
        comps=bom.GetComponents2(row,config)
        if comps:
            counts.append({'row':row,'count':len(comps),'paths':[get(c,'GetPathName') for c in comps]})
    return cells,counts
