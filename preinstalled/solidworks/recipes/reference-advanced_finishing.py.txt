def open_copy(relative,name):
    source=(ROOT/relative).resolve();assert source.is_relative_to(ROOT.resolve()) and source.is_file()
    dest=r.out/(r.out.name+'-'+name);assert not dest.exists();shutil.copy2(source,dest)
    e,w=VARIANT(pythoncom.VT_BYREF|pythoncom.VT_I4,0),VARIANT(pythoncom.VT_BYREF|pythoncom.VT_I4,0)
    d=r.app.OpenDoc6(str(dest),1,1,'',e,w)
    r.check('open_isolated_copy',d is not None and e.value==0)
    assert Path(get(d,'GetPathName')).resolve()==dest
    r.owned.add(get(d,'GetTitle'));return d
def properties(feat):
    mgr=get(feat,'CustomPropertyManager');values={}
    for n in get(mgr,'GetNames') or []:
        dispid=mgr._oleobj_.GetIDsOfNames('Get2')
        raw,value=mgr._oleobj_.InvokeTypes(dispid,0,1,(24,0),((8,1),(16392,2),(16392,2)),n)
        values[n]={'raw':raw,'value':value,'method':'SW2016 Get2 typed out strings'}
    return values
